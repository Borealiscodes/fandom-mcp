\section*{Overview}

This document defines the A7.3 reprojection operators responsible for
mapping repaired collapse-surface geometry $\Sigma_{\mathrm{repaired}}$
back into a dual-compatible manifold $M_{\mathrm{A6}}$.

A7.3 operators must:
\begin{itemize}
  \item restore dual algebraic structure,
  \item preserve repaired invariants,
  \item maintain continuity and adjacency,
  \item reestablish spectral stability,
  \item enforce holonomy bounds,
  \item preserve kernel invariants.
\end{itemize}

A7.3 is the final altitude before returning to A6-compatible geometry.

% ------------------------------------------------------------
\section{Reprojection Operator Definition}

Define the reprojection operator:
\[
\mathcal{P}_{\mathrm{A7.3}} :
\Sigma_{\mathrm{repaired}} \to M_{\mathrm{A6}}.
\]

Reprojection must satisfy:
\[
\mathcal{P}_{\mathrm{A7.3}}(p) = p
\quad \text{if } p \in M_{\mathrm{A6}}.
\]

Reprojection is idempotent:
\[
\mathcal{P}_{\mathrm{A7.3}}^2 = \mathcal{P}_{\mathrm{A7.3}}.
\]

% ------------------------------------------------------------
\section{Continuity Restoration}

Define continuity restoration:
\[
\mathcal{P}_{\mathrm{cont}}(p)
  = \lim_{\epsilon \to 0} B_\epsilon(p).
\]

Reprojection must satisfy:
\[
p_{\mathrm{reproj}} \in C^0.
\]

% ------------------------------------------------------------
\section{Adjacency Restoration}

Define adjacency restoration:
\[
\mathcal{P}_{\mathrm{adj}}(p)
  = \{ q \in M_{\mathrm{A6}} : d(p,q) < \epsilon_{\mathrm{adj}} \}.
\]

Reprojection must satisfy:
\[
\mathrm{Adj}(p_{\mathrm{reproj}}) = \mathrm{Adj}_{\mathrm{A6}}(p).
\]

% ------------------------------------------------------------
\section{Spectral Stability Restoration}

Define spectral reprojection:
\[
\mathcal{P}_{\mathrm{spec}}(p)
  = \mathrm{Spec}_{\mathrm{A6}}(p).
\]

Reprojection must satisfy:
\[
\delta_{\mathrm{spec}}(p_{\mathrm{reproj}}) < \epsilon_{\mathrm{stable}}.
\]

% ------------------------------------------------------------
\section{Holonomy Restoration}

Define holonomy reprojection:
\[
\mathcal{P}_{\mathrm{hol}}(\gamma)

\]

Reprojection must satisfy:
\[
H(p_{\mathrm{reproj}}) < \Theta_{\max}.
\]

% ------------------------------------------------------------
\section{Kernel Reintegration}

Define kernel reintegration:
\[
\mathcal{P}_{\mathrm{kernel}}(p)
  = K_{\mathrm{A6}}(p).
\]

Reprojection must satisfy:
\[
K(p_{\mathrm{reproj}}) = K_{\mathrm{A6}}(p).
\]

This ensures dual algebra compatibility.

\[
\mathcal{P}_{\mathrm{A7.3}}(p)
  = \mathcal{P}_{\mathrm{cont}}
    \circ \mathcal{P}_{\mathrm{adj}}

\[
p_{\mathrm{reproj}} \in M_{\mathrm{A6}},
\]
\[
\Gamma_{\mathrm{reproj}}(p) = \Gamma_{\mathrm{A6}}(p).
\]

% ------------------------------------------------------------
