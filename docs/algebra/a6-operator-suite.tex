\section{Curvature Operators}

\subsection{Curvature Lift Operator}
Given curvature field $K(p)$, define the lift:
\[
\mathcal{C}_{\mathrm{lift}}(K)(p) = K(p) + \delta K(p),
\]
subject to curvature continuity:
\[
|\delta K(p)| < \epsilon.
\]

\subsection{Curvature Combination Operator}

\[
\mathcal{C}_{\mathrm{comb}}(K_A, K_B)
  = \sqrt{K_A^2 + K_B^2 + 2 K_A K_B \Phi(A,B)}.
\]
